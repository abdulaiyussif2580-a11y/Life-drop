// Example SQL Query: Find O- Donors Phone and Email in Takoradi
SELECT d.full_name, d.phone_number, d.email_address, d.city_location
FROM tbl_donors d
WHERE d.city_location = 'Takoradi'
AND d.blood_group = 'O-'
AND d.status = 'AVAILABLE';
